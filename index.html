Como atribuir as notas de 0 a 10? (Referência para o Avaliador)

Converse com o grupo e escolha a nota que melhor representa a experiência real de um pescador ou marisqueira ao utilizar este canal:

0 a 3 • Crítico / Inadequado Canal inacessível no celular, linguagem técnica excludente ("de gabinete") ou que não oferece dados úteis para fiscalizar.
4 a 6 • Regular / Parcial A informação existe, mas exige esforço para encontrar, mistura termos difíceis ou divulga ações apenas depois que já ocorreram.
7 a 8 • Bom / Adequado Fácil de navegar, respeita o vocabulário da pesca artesanal e traz informações úteis para acompanhar o andamento do projeto.
9 a 10 • Excelente / Referência Leve no celular, linguagem popular exemplar, com transparência ativa que fortalece a decisão e a fiscalização pela comunidade.

Integração em Tempo Real com Google Sheets (Sincronização 60s)

Conecte todos os grupos à mesma planilha Google Sheets para atualização mútua de minuto em minuto.

* Se deixado em branco, a ferramenta funciona em modo de demonstração local e sincroniza entre abas do mesmo navegador. Com a URL do Apps Script ativa, todos os computadores e celulares dos 10 grupos compartilham a mesma base a cada 60 segundos.

Código Google Apps Script (Copie e cole em Extensões > Apps Script na sua Planilha)
  1. Crie uma Planilha Google em branco e clique em Extensões > Apps Script.
  2. Apague tudo, cole o código abaixo e clique em Implantar > Nova implantação > App da Web.
  3. Em "Quem pode acessar", selecione "Qualquer pessoa", copie a URL gerada e cole no campo acima.
function doGet(e) {
  var sheet = getOrCreateSheet_();
  var data = sheet.getDataRange().getValues();
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) {
      rows.push({
        id: Number(data[i][0]),
        reporter: data[i][1] || "",
        municipalities: data[i][2] || "",
        acc: Number(data[i][3] || 5),
        lang: Number(data[i][4] || 5),
        util: Number(data[i][5] || 5),
        selectedTags: data[i][6] ? JSON.parse(data[i][6]) : [],
        customTags: data[i][7] ? JSON.parse(data[i][7]) : [],
        bottlenecks: data[i][8] || "",
        proposal: data[i][9] || "",
        updatedAt: data[i][10] || "",
        completed: Boolean(data[i][11])
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", groups: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var payload = JSON.parse(e.postData.contents);
    var sheet = getOrCreateSheet_();
    var group = payload.group;
    var rowIndex = group.id + 1;
    sheet.getRange(rowIndex, 1, 1, 12).setValues([[
      group.id,
      group.reporter,
      group.municipalities,
      group.acc,
      group.lang,
      group.util,
      JSON.stringify(group.selectedTags || []),
      JSON.stringify(group.customTags || []),
      group.bottlenecks,
      group.proposal,
      new Date().toLocaleTimeString("pt-BR"),
      true
    ]]);
    return ContentService.createTextOutput(JSON.stringify({ status: "saved" }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("ControleSocial_Grupos");
  if (!sheet) {
    sheet = ss.insertSheet("ControleSocial_Grupos");
    sheet.appendRow(["Grupo_ID","Relator","Municipios","Acessibilidade","Linguagem","Utilidade","Tags_Selecionadas","Tags_Criadas","Gargalos","Proposta_Carta","Atualizado_Em","Concluido"]);
    for (var i = 1; i <= 10; i++) {
      sheet.appendRow([i, "", "", 5, 5, 5, "[]", "[]", "", "", "", false]);
    }
  }
  return sheet;
}
